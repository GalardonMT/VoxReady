### FASE 1: MÓDULO GENERADOR NARRATIVO (`core/narrative_generator.py`)

Crear el componente encargado de la segunda llamada a la API de NVIDIA Build (`meta/llama-3.2-90b-vision-instruct`), parametrizado para emular la estructura de evaluación, el tono diplomático y las herramientas verbales del estándar VISUM.

Crear el archivo `voxready-worker/core/narrative_generator.py`:

```python
import json
import logging
from typing import Dict, Any
from openai import OpenAI

logger = logging.getLogger(__name__)

SYSTEM_PROMPT_VISUM = """Eres un consultor senior de comunicaciones estratégicas y media training de VISUM Consulting.
Tu tarea es redactar el Informe Ejecutivo de Evaluación de Vocería a partir de los datos consolidados de una simulación de crisis.

PRINCIPIOS DE REDACCIÓN Y METODOLOGÍA VISUM:
1. Tono ejecutivo, riguroso, diplomático y pedagógico: "Comunicar bien no es un talento innato: es una disciplina que se entrena".
2. Estructura balanceada: valida el piso sólido de competencias del vocero antes de señalar oportunidades de desarrollo avanzadas.
3. Traduce métricas técnicas a impacto comunicacional:
   - Contacto visual y postura -> Aplomo no verbal, compostura y seguridad ante el público.
   - WPM, pausas y fonación -> Control emocional, ritmo pedagógico y claridad expositiva.
   - Apego al mensaje y bridging -> Disciplina informativa y conducción estratégica.
   - Señal cruzada -> Congruencia entre el discurso verbal y la corporalidad/gesto.
4. Fórmulas prácticas obligatorias:
   - Entrevista técnica -> Formulación pedagógica: "Para que todas las personas lo entiendan, este tema se compone de..."
   - Interacción off the record -> Pregunta de preparación: "¿Qué debería comprender mejor este periodista después de conversar conmigo?"
   - Crisis operacional -> Secuencia en 3 pasos: Reconocer el impacto -> Explicar certezas -> Acciones en curso.

ESQUEMA JSON DE RESPUESTA REQUERIDO:
Responde EXCLUSIVAMENTE un objeto JSON válido con la siguiente estructura:
{
  "sintesis_ejecutiva": {
    "diagnostico_general": "Párrafo de 3 a 4 líneas que evalúe el desempeño global en la simulación.",
    "fortaleza_principal": "Piso sólido de competencias demostrado.",
    "foco_desarrollo": "Evolución requerida (de responder correctamente a conducir la conversación).",
    "continuidad_recomendada": "Recomendación de entrenamiento continuo adaptada al escenario."
  },
  "desempeno_observado": {
    "evaluacion_general": "Análisis contextualizado de la interacción, autocontrol y manejo de la presión.",
    "fortalezas": [
      "Fortaleza concreta 1",
      "Fortaleza concreta 2",
      "Fortaleza concreta 3"
    ],
    "oportunidades_desarrollo": [
      "Oportunidad accionable 1",
      "Oportunidad accionable 2",
      "Oportunidad accionable 3"
    ],
    "formula_practica_recomendada": "Formulación verbal o secuencia aplicable al tipo de crisis practicado."
  },
  "hallazgos_transversales": [
    {
      "prioridad": "Pedagogía de la complejidad | Conducción estratégica | Empatía como primera señal | Delimitación institucional",
      "descripcion": "Explicación de la brecha y su impacto estratégico en la audiencia."
    }
  ],
  "recomendaciones_proximas_vocerias": [
    "Regla de oro accionable 1",
    "Regla de oro accionable 2",
    "Regla de oro accionable 3",
    "Regla de oro accionable 4"
  ],
  "consideracion_final": "Párrafo de cierre sobre la práctica y el perfeccionamiento de habilidades.",
  "observacion_senal_cruzada": "Cita o momento exacto donde la corporalidad, la voz o el mensaje tuvieron sinergia o contradicción evidente."
}
"""

def generate_visum_report(
    client: OpenAI,
    model_name: str,
    scenario_info: Dict[str, Any],
    consolidated_json: Dict[str, Any]
) -> Dict[str, Any]:
    """
    Invoca a NVIDIA NIM en una llamada independiente para generar la narrativa VISUM.
    """
    prompt_payload = {
        "escenario": {
            "titulo": scenario_info.get("title"),
            "contexto": scenario_info.get("context"),
            "optica": scenario_info.get("optics"),
            "mensajes_clave": scenario_info.get("key_messages", []),
            "lineas_rojas": scenario_info.get("red_lines", [])
        },
        "metricas_consolidadas": {
            "score_global": consolidated_json.get("score_global"),
            "expresion_no_verbal": consolidated_json.get("evaluacion_areas", {}).get("expresion"),
            "tono_de_voz": consolidated_json.get("evaluacion_areas", {}).get("tono_voz"),
            "contenido_estrategico": consolidated_json.get("evaluacion_areas", {}).get("contenido"),
            "empatia_congruencia": consolidated_json.get("evaluacion_areas", {}).get("empatia")
        },
        "transcripcion": consolidated_json.get("transcript", "")
    }

    user_message = f"Genera el informe ejecutivo VISUM para el siguiente desempeño evaluado:\n\n{json.dumps(prompt_payload, ensure_ascii=False, indent=2)}"

    try:
        response = client.chat.completions.create(
            model=model_name,
            messages=[
                {"role": "system", "content": SYSTEM_PROMPT_VISUM},
                {"role": "user", "content": user_message}
            ],
            temperature=0.5,
            response_format={"type": "json_object"}
        )
        return json.loads(response.choices[0].message.content)
    except Exception as e:
        logger.error(f"Fallo en la generación del informe VISUM: {str(e)}")
        raise

```

### FASE 2: ORQUESTACIÓN Y PERSISTENCIA TRANSACCIONAL EN SQL

Modificar la etapa de consolidación en el worker (`core/pipeline_orchestrator.py` o `main.py`) para encadenar la fusión matemática, la generación VISUM y el guardado en base de datos bajo una sola transacción ODBC.

```python
import uuid
import json
import pyodbc
from core.narrative_generator import generate_visum_report

def finalize_session_analysis(
    db_conn: pyodbc.Connection,
    nvidia_client,
    model_name: str,
    session_id: str,
    job_id: str,
    rubric_version_id: str,
    scenario_metadata: dict,
    consolidated_scores: dict
):
    """
    1. Genera la narrativa ejecutiva mediante llamada independiente al LLM.
    2. Persiste atomicamente en las tablas report, area_score, analysis_pipeline_result.
    3. Actualiza los estados de analysis_job y session a 'completed'.
    """
    cursor = db_conn.cursor()

    try:
        # Paso 1: Generación de narrativa desacoplada (Llamada 2 a NVIDIA NIM)
        visum_narrative = generate_visum_report(
            client=nvidia_client,
            model_name=model_name,
            scenario_info=scenario_metadata,
            consolidated_json=consolidated_scores
        )

        # Paso 2: Crear el registro de informe principal (report)
        report_id = str(uuid.uuid4())
        overall_score = int(round(consolidated_scores["score_global"]))
        strengths_json = json.dumps(visum_narrative["desempeno_observado"]["fortalezas"], ensure_ascii=False)
        improvements_json = json.dumps(visum_narrative["desempeno_observado"]["oportunidades_desarrollo"], ensure_ascii=False)
        cross_signal_obs = visum_narrative.get("observacion_senal_cruzada", "")[:1000]

        cursor.execute("""
            INSERT INTO report (
                id, session_id, analysis_job_id, rubric_version_id, overall_score,
                narrative_strengths, narrative_improvements, cross_signal_observation,
                version_no, is_current, generated_at, created_at, updated_at
            )
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, 1, 1, SYSUTCDATETIME(), SYSUTCDATETIME(), SYSUTCDATETIME())
        """, (
            report_id, session_id, job_id, rubric_version_id, overall_score,
            strengths_json, improvements_json, cross_signal_obs
        ))

        # Paso 3: Persistir los puntajes por área (area_score)
        # Resuelve las FK dinámicas de rubric_area vinculadas a la versión de rúbrica activa
        cursor.execute("""
            SELECT id, area_key FROM rubric_area WHERE rubric_version_id = ?
        """, rubric_version_id)
        rubric_areas = {row.area_key: row.id for row in cursor.fetchall()}

        scores_map = {
            "expression": int(round(consolidated_scores["evaluacion_areas"]["expresion"]["score"])),
            "voice": int(round(consolidated_scores["evaluacion_areas"]["tono_voz"]["score"])),
            "coherence": int(round(consolidated_scores["evaluacion_areas"]["contenido"]["score"])),
            "empathy": int(round(consolidated_scores["evaluacion_areas"]["empatia"]["score"]))
        }

        for area_key, area_val in scores_map.items():
            rubric_area_id = rubric_areas.get(area_key)
            if rubric_area_id:
                cursor.execute("""
                    INSERT INTO area_score (
                        id, report_id, rubric_area_id, value, created_at, updated_at
                    )
                    VALUES (NEWID(), ?, ?, ?, SYSUTCDATETIME(), SYSUTCDATETIME())
                """, (report_id, rubric_area_id, area_val))

        # Paso 4: Guardar payload completo en analysis_pipeline_result (pipeline = 'fusion')
        full_summary_payload = {
            "scores": consolidated_scores,
            "visum_report": visum_narrative
        }
        cursor.execute("""
            INSERT INTO analysis_pipeline_result (
                id, analysis_job_id, pipeline, status, summary_json, created_at, updated_at
            )
            VALUES (NEWID(), ?, 'fusion', 'done', ?, SYSUTCDATETIME(), SYSUTCDATETIME())
        """, (job_id, json.dumps(full_summary_payload, ensure_ascii=False)))

        # Paso 5: Cerrar máquina de estados en analysis_job y session
        cursor.execute("""
            UPDATE analysis_job
            SET status = 'completed', completed_at = SYSUTCDATETIME(), updated_at = SYSUTCDATETIME()
            WHERE id = ?
        """, job_id)

        cursor.execute("""
            UPDATE session
            SET status = 'completed', updated_at = SYSUTCDATETIME()
            WHERE id = ?
        """, session_id)

        db_conn.commit()
        logger.info(f"Sesión {session_id} completada y persistida exitosamente con Report ID: {report_id}")

    except Exception as ex:
        db_conn.rollback()
        logger.error(f"Error persistiendo informe para sesión {session_id}: {str(ex)}")
        cursor.execute("""
            UPDATE analysis_job
            SET status = 'failed', failure_reason = ?, updated_at = SYSUTCDATETIME()
            WHERE id = ?
        """, (str(ex)[:500], job_id))
        db_conn.commit()
        raise
    finally:
        cursor.close()

```
